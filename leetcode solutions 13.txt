5. longest palindromic substring


class Solution {

    // Главный метод, который получает строку и должен вернуть
    // самую длинную палиндромную подстроку
    public String longestPalindrome(String s) {

        // Если строка состоит из 0 или 1 символа,
        // она уже является палиндромом
        if (s.length() < 2) {
            return s;
        }

        // Здесь будем хранить начало самого длинного палиндрома
        int start = 0;

        // Здесь будем хранить конец самого длинного палиндрома
        int end = 0;

        // Перебираем каждый символ строки и рассматриваем его как центр палиндрома
        for (int i = 0; i < s.length(); i++) {

            // Проверяем палиндром с центром прямо на символе i
            // Например: "aba"
            int len1 = expandAroundCenter(s, i, i);

            // Проверяем палиндром с центром между i и i + 1
            // Например: "abba"
            int len2 = expandAroundCenter(s, i, i + 1);

            // Берём длину самого большого найденного палиндрома
            int len = Math.max(len1, len2);

            // Если найденный палиндром длиннее нашего предыдущего ответа
            if (len > end - start + 1) {

                // Вычисляем левую границу нового палиндрома
                start = i - (len - 1) / 2;

                // Вычисляем правую границу нового палиндрома
                end = i + len / 2;
            }
        }

        // Возвращаем часть строки от start до end включительно
        // В substring правая граница не включается,
        // поэтому пишем end + 1
        return s.substring(start, end + 1);
    }

    // Этот метод расширяется от центра влево и вправо
    // и проверяет, является ли полученная часть палиндромом
    private int expandAroundCenter(String s, int left, int right) {

        // Пока left находится внутри строки,
        // right находится внутри строки,
        // и символы слева и справа одинаковые,
        // продолжаем расширяться
        while (left >= 0 && right < s.length()
                && s.charAt(left) == s.charAt(right)) {

            // Двигаемся на один символ влево
            left--;

            // Двигаемся на один символ вправо
            right++;
        }

        // После выхода из цикла left и right уже находятся
        // на позициях за пределами палиндрома
        // Поэтому длина палиндрома равна right - left - 1
        return right - left - 1;
    }
}
